    def normalize_article_text(raw):
        if not isinstance(raw, str):

        raw = html.unescape(raw)
        if re.search(r"<[a-zA-Z][^>]*>", raw):
            raw = BeautifulSoup(raw, "html.parser").get_text(" ", strip=True)
        for ch in _INVISIBLE:

        raw = re.sub(r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b", " ", raw)
        return re.sub(r"\s+", " ", raw).strip()

    def has_balanced_brackets(text):
        pairs = {"(": ")", "[": "]", "{": "}"}
        stack = []
        for ch in text:
            if ch in pairs:
                stack.append(ch)
            elif ch in pairs.values():
                if not stack or pairs[stack.pop()] != ch:
                    return False
        return not stack

    enc = tokenizer(batch["input"], max_length=SRC_MAX_LEN, truncation=True)

# 12. Metrics
# ============================================================
_MAIN_RE = re.compile(r"<main_class>\s*(.*?)\s*<sub_class>", re.DOTALL)
_SUB_RE  = re.compile(r"<sub_class>\s*(.*?)(?:</s>|$)",      re.DOTALL)

def parse_label_tokens(text):
    text = text.replace("<pad>", "").strip()
    m = _MAIN_RE.search(text)
    s = _SUB_RE.search(text)
    return (m.group(1).strip() if m else "",
            s.group(1).strip() if s else "")

def seq2seq_metrics(eval_pred):
    preds, labels = eval_pred
    if isinstance(preds, tuple):
        preds = preds[0]
    decoded_preds = tokenizer.batch_decode(preds, skip_special_tokens=False)
    labels = np.where(labels != -100, labels, tokenizer.pad_token_id)
    decoded_labels = tokenizer.batch_decode(labels, skip_special_tokens=False)
    total = len(decoded_labels)
    main_ok = sub_ok = exact_ok = 0
    for p, t in zip(decoded_preds, decoded_labels):
        pm, ps = parse_label_tokens(p)
        tm, ts = parse_label_tokens(t)
        if pm == tm: main_ok += 1
        if ps == ts: sub_ok  += 1
        if pm == tm and ps == ts: exact_ok += 1
    return {
        "main_acc": round(main_ok/total, 4),
        "sub_acc" : round(sub_ok/total, 4),
        "exact"   : round(exact_ok/total, 4),
    }

def predict_article(title, content, num_beams=4):
    prompt = ("المهمة: تصنيف هرمي\n\n"
              f"العنوان:\n{title}\n\n"
              f"النص:\n{content}")
    enc = tokenizer(prompt, max_length=SRC_MAX_LEN,
                    truncation=True, return_tensors="pt").to(model.device)
    with torch.no_grad():
        out = model.generate(**enc, max_length=TGT_MAX_LEN,
                             num_beams=num_beams, early_stopping=True)
    raw = tokenizer.decode(out[0], skip_special_tokens=False)
    m, s = parse_label_tokens(raw)
    return {"main": m, "sub": s, "raw": raw}


def predict_batch(items, batch_size=8):
    results = []
    for i in range(0, len(items), batch_size):
        chunk = items[i:i + batch_size]
        prompts = [
            "المهمة: تصنيف هرمي\n\n"
            f"العنوان:\n{c['title']}\n\n"
            f"النص:\n{c['content']}"
            for c in chunk
        ]
        enc = tokenizer(prompts, max_length=SRC_MAX_LEN,
                        truncation=True, padding=True,
                        return_tensors="pt").to(model.device)
        with torch.no_grad():
            out = model.generate(**enc, max_length=TGT_MAX_LEN, num_beams=1)
        for text in tokenizer.batch_decode(out, skip_special_tokens=False):
            m, s = parse_label_tokens(text)
            results.append({"main": m, "sub": s})
    return results

        print(f"  {f}  ({os.path.getsize(p)/1e6:.1f} MB)")
