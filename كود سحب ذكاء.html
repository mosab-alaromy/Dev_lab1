✅ الطريقة غير المتزامنة

🔒 Semaphore — التحكم في التزامن

semaphore = asyncio.Semaphore(35)

async def fetch_article(session, url, semaphore):
    async with semaphore:
        # لا يمكن تنفيذ أكثر من 35 طلباً في نفس اللحظة
        async with session.get(url, ...) as resp:
            ...
🎯
لماذا Semaphore مهم؟ يمنع فتح عدد لا نهائي من الاتصالات الذي يؤدي إلى حظر السيرفر لنا (Rate Limiting) أو استنزاف الذاكرة.

🚀 تشغيل النظام بشكل دفعات

for i in range(0, len(all_urls), 300):
    chunk = all_urls[i:i+300]
    tasks = [fetch_article(session, u, semaphore) for u in chunk]

🛡️
تم استخدام return_exceptions=True لضمان عدم توقف باقي المهام حتى لو فشلت إحداها — تُعاد الأخطاء كقيم بدل أن تُرفع كاستثناءات.

🔁 Retry + Exponential Backoff (جديد)

for attempt in range(MAX_RETRIES):          # 3 محاولات
    try:
        async with session.get(url, headers=headers,
                timeout=aiohttp.ClientTimeout(total=10)) as resp:

            if resp.status == 429:              # Rate-Limit
                await asyncio.sleep(3 * (attempt + 1))
                continue

            if resp.status != 200:
                return None

            ctype = resp.headers.get('Content-Type', '')
            if 'text/html' not in ctype:        # فحص نوع المحتوى
                return None

            html = await resp.text(encoding='utf-8', errors='ignore')
            return extract_article_data(url, html)

    except asyncio.TimeoutError:
        ERROR_STATS['timeout'] += 1
        await asyncio.sleep(1.5 ** attempt)     # Backoff تصاعدي
    except aiohttp.ClientError as e:
        ERROR_STATS[f'client:{type(e).__name__}'] += 1
        await asyncio.sleep(1.5 ** attempt)

معالجة 429 (Rate-Limit)

عند حظر مؤقت، ينتظر النظام 3×(محاولة+1) ثانية ثم يعيد المحاولة تلقائياً.

التحقق من Content-Type

تجاهل أي استجابة ليست text/html (مثل الصور أو JSON) لتوفير الوقت.

6 تنظيف واستخلاص النص

🧹 دالة تنظيف النص

def clean_text(text):
    if not text:
        return ""
    return re.sub(r'\s+', ' ', text).strip()

تقوم الدالة بتوحيد جميع المسافات المتعددة (مسافات، أسطر جديدة، Tabs) إلى مسافة واحدة، وإزالة الفراغات من الأطراف.

💾 الحفظ بالدفعات (Batch Insert)

def save_batch_to_db(batch):
    if not batch:
        return 0
    conn = sqlite3.connect(LOCAL_DB_FILE)
    cursor = conn.cursor()
    cursor.executemany(
        "INSERT OR IGNORE INTO articles VALUES (?, ?, ?, ?, ?)",
        batch
    )
    inserted = cursor.rowcount
    conn.commit()
    conn.close()
    return inserted
⚡
INSERT OR IGNORE: يحاول الإدراج، وإن وُجد الرابط مسبقاً (Primary Key) يتجاهله بصمت — حماية مزدوجة من التكرار.

def load_existing_urls():
    saved_urls = set()

    # 1) من ملف JSONL على Drive  (سطر لكل مقال — سريع وآمن)
    if os.path.exists(JSONL_FILE):
        with open(JSONL_FILE, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if not line: continue
                try:
                    obj = json.loads(line)
                    if 'url' in obj:
                        saved_urls.add(obj['url'])
                except json.JSONDecodeError:
                    ERROR_STATS['json_lines_corrupted'] += 1

    # 2) من قاعدة البيانات المحلية
    if os.path.exists(LOCAL_DB_FILE):
        conn = sqlite3.connect(LOCAL_DB_FILE)
        cursor = conn.cursor()
        cursor.execute("SELECT url FROM articles")
        for (url,) in cursor.fetchall():
            saved_urls.add(url)
        conn.close()

    return saved_urls
🆕
التحوّل من JSON Array إلى JSONL: صار كل مقال في سطر مستقل، فلا داعي لتخزين الملف كاملاً في الذاكرة، وأصبح التصدير والاسترجاع أسرع وأكثر أماناً ضد التلف.

10 التصدير والمزامنة (محدَّث)

📤 تصدير JSONL بدل JSON (تحسين جوهري)

def export_to_jsonl_and_sync():
    conn = sqlite3.connect(LOCAL_DB_FILE)
    cursor = conn.cursor()
    cursor.execute(
        "SELECT url, main_category, sub_category, text, scraped_at "
        "FROM articles"
    )

    tmp_file = JSONL_FILE + '.tmp'
    count = 0
    with open(tmp_file, 'w', encoding='utf-8') as f:
        while True:
            rows = cursor.fetchmany(5000)      # تصدير بالدفعات
            if not rows: break
            for row in rows:
                record = {
                    "url"          : row[0],
                    "main_category": row[1] or "",
                    "sub_category" : row[2] or "",
                    "text"         : row[3],
                    "scraped_at"   : row[4] or ""
                }
                f.write(json.dumps(record, ensure_ascii=False) + "\n")
                count += 1
    conn.close()

    # 🆕 استبدال ذرّي (Atomic Replace)
    os.replace(tmp_file, JSONL_FILE)

    # نسخ قاعدة البيانات إلى Drive
    shutil.copy2(LOCAL_DB_FILE, DRIVE_DB_FILE)

🛡️ لماذا الاستبدال الذرّي؟

نكتب أولاً إلى ملف مؤقت .tmp، ثم نستبدل الأصلي عبر os.replace. فلو انقطعت الجلسة أثناء الكتابة، لن يتلف الملف الأصلي — إما نسخة قديمة كاملة أو نسخة جديدة كاملة.

⏰ متى تتم المزامنة؟ (تحديث)

دورياً كل 5,000 مقال

async def main():
    init_db()                                # تهيئة الجدول
    saved_urls = load_existing_urls()
    current_saved = len(saved_urls)

    if current_saved >= TARGET_ARTICLES:
        print("🎉 الهدف محقق مسبقاً")
        export_to_jsonl_and_sync()
        return

    semaphore = asyncio.Semaphore(CONCURRENCY_LIMIT)   # 35
    connector = aiohttp.TCPConnector(limit=TCP_LIMIT, ssl=False)  # 70

    async with aiohttp.ClientSession(connector=connector) as session:
        start_id = await get_latest_article_id(session)   # 🆕 ديناميكي
        end_id   = 100_000
        needed   = (TARGET_ARTICLES - current_saved) * 2  # 🆕 2×

        all_urls = [f"https://www.sarayanews.com/article/{aid}"
                    for aid in range(start_id, end_id, -1)
                    if f"https://www.sarayanews.com/article/{aid}"
                        not in saved_urls][:needed]

        batch = []
        last_export = current_saved
        for i in range(0, len(all_urls), BATCH_SIZE):
            chunk = all_urls[i:i+BATCH_SIZE]
            tasks = [fetch_article(session, u, semaphore) for u in chunk]
            results = await asyncio.gather(*tasks, return_exceptions=True)

            for res in results:
                if isinstance(res, tuple):
                    batch.append(res)

            if batch:
                inserted = save_batch_to_db(batch)
                current_saved += inserted
                batch.clear()

            # 🆕 مزامنة كل 5,000 مقال
            if current_saved - last_export >= EXPORT_EVERY:
                export_to_jsonl_and_sync()
                last_export = current_saved

            if current_saved >= TARGET_ARTICLES:
                break

    export_to_jsonl_and_sync()      # تصدير نهائي
    # 🆕 طباعة إحصائيات الأخطاء
    for k, v in ERROR_STATS.most_common():
        print(f"   {k}: {v:,}")

13 آليات المتانة (بعد التعديل)

🛡️ 10 طبقات حماية (بدل 8)

1. الاستئناف التلقائي

استرجاع الروابط من DB + JSONL عند إعادة التشغيل.

2. منع التكرار المزدوج

PRIMARY KEY + INSERT OR IGNORE + set() في الذاكرة.

3. 🆕 Retry ذكي (3 محاولات)

يعيد المحاولة تلقائياً عند فشل الشبكة مع Backoff تصاعدي.

4. Timeout 10s

لا يعلق النظام على أي طلب — مهلة أعلى لشبكات أبطأ.

5. 🆕 معالجة 429

يتعامل مع Rate-Limit بالانتظار بدل الفشل.

6. Semaphore 35

يمنع سحق السيرفر أو حظر الـIP.

7. User-Agent Rotation

محاكاة متصفحات حقيقية لتفادي الكشف.

8. 🆕 كتابة ذرّية (Atomic)

ملف .tmp ثم os.replace — يستحيل تلف JSONL.

9. فحوصات الجودة

رفض المقالات القصيرة، صفحـات الخطأ، وContent-Type غير HTML.

10. 🆕 إحصائيات الأخطاء

ERROR_STATS = Counter() لتشخيص نوع الفشل بدقة.
